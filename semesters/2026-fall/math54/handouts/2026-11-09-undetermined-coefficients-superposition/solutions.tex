\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P1}{P1}.}\par\idea{$p(r)=r^2-r-2=(r-2)(r+1)$, so $0$ is not a root ($s=0$): try a linear polynomial $At+B$.}
  \begin{steps}
    \ii With $y=At+B$, $y'=A$, $y''=0$, so $y''-y'-2y=-2At+(-A-2B)$.
    \ii Matching $4t$: $-2A=4$ and $-A-2B=0\implies A=-2,\ B=1$, hence $y_p=-2t+1$.
    \ii Check: $0-(-2)-2(-2t+1)=4t$.
  \end{steps}
\end{prob}

\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P2}{P2}.}\par\idea{$p(r)=r^2+2r+5$ has roots $-1\pm 2i$, so $i$ is not a root ($s=0$); the complex method replaces a $2\times 2$ system by one complex division.}
  \begin{steps}
    \ii \textbf{Real coefficients:} Trying $y=A\cos t+B\sin t$ gives $y'=-A\sin t+B\cos t$, $y''=-A\cos t-B\sin t$, so
    \[
      y''+2y'+5y = (4A+2B)\cos t + (4B-2A)\sin t = 10\cos t.
    \]
    Hence $4A+2B=10$, $-2A+4B=0\implies A=2B$, $10B=10\implies B=1,\ A=2$: $y_p=2\cos t+\sin t$.
    \ii \textbf{Complex exponential:} $p(i)=-1+2i+5=4+2i$, so $z=\frac{10}{4+2i}e^{it}$ solves $L[z]=10e^{it}$. Since
    \[
      \frac{10}{4+2i}=\frac{10(4-2i)}{20}=2-i,
    \]
    \[
      y_p = \operatorname{Re}z = \operatorname{Re}\bigl((2-i)(\cos t+i\sin t)\bigr) = 2\cos t+\sin t.
    \]
  \end{steps}
\end{prob}

\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P3}{P3}.}\par\idea{$2$ is a simple root of $p(r)=r^2-4$ ($s=1$); use \xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prop-key-formula}{the key product formula}.}
  \begin{steps}
    \ii $p(2)=0$ and $p'(2)=4\neq 0$, so $L[te^{2t}]=p'(2)e^{2t}=4e^{2t}$.
    \ii Hence $y_p=\frac84 te^{2t}=2te^{2t}$. Check: $y_p'=(2+4t)e^{2t}$, $y_p''=(8+8t)e^{2t}$, $y_p''-4y_p=8e^{2t}$.
  \end{steps}
\end{prob}

\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P4}{P4}.}\par\idea{$-3$ is a double root of $p(r)=(r+3)^2$ ($s=2$), so $L[qe^{-3t}]=q''e^{-3t}$.}
  \begin{steps}
    \ii Setting $q''=2$ gives $q=t^2$ and $y_p=t^2e^{-3t}$.
    \ii Check: $y_p'=(2t-3t^2)e^{-3t}$, $y_p''=(2-12t+9t^2)e^{-3t}$, giving $y_p''+6y_p'+9y_p=2e^{-3t}$.
  \end{steps}
\end{prob}

\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P5}{P5}.}\par\idea{$p(r)=r^2+4$ has roots $\pm 2i$; compare $2i$, $2+i$, and $2$ against $\pm 2i$ (\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{name-undetermined-revisited}{Nagle 4.5}).}
  \begin{steps}
    \ii \textbf{(a) $t\sin 2t$:} $\alpha+i\beta=2i$ is a root ($s=1$, $m=1$), so $y_p=t(A_1t+A_0)\cos 2t+t(B_1t+B_0)\sin 2t$.
    \ii \textbf{(b) $e^{2t}\cos t$:} $2+i$ is not a root ($s=0$), so $y_p=Ae^{2t}\cos t+Be^{2t}\sin t$.
    \ii \textbf{(c) $3t^2e^{2t}+\cos 2t$:} $2$ is not a root ($s=0$), while $2i$ is ($s=1$). By \xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{thm-nagle-4-3}{Superposition},
    \[
      y_p = (A_2t^2+A_1t+A_0)e^{2t} + t(C\cos 2t+D\sin 2t).
    \]
  \end{steps}
\end{prob}

\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P6}{P6}.}\par\idea{Since $6t-4e^t=3(2t)-4(e^t)$, \xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{thm-nagle-4-3}{Superposition} yields $y_p$ directly, and \xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{thm-nagle-4-4}{Nagle Thm.~4} adds $y_h$.}
  \begin{steps}
    \ii By superposition,
    \[
      y_p = 3y_1-4y_2 = 3\bigl(-t+\tfrac12\bigr)-4\bigl(-\tfrac12 e^t\bigr) = -3t+\tfrac32+2e^t.
    \]
    \ii $r^2-r-2=(r-2)(r+1)\implies y_h=c_1e^{2t}+c_2e^{-t}$, so all solutions are
    \[
      y = -3t + \frac32 + 2e^t + c_1e^{2t} + c_2e^{-t}.
    \]
  \end{steps}
\end{prob}

\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P7}{P7}.}\par\idea{Find $y_p$, add $y_h$, and apply initial conditions to the total sum $y=y_p+y_h$.}
  \begin{steps}
    \ii Trying $y_p=At+B$ in $y''-y=2t$ gives $-At-B=2t\implies A=-2, B=0$, so $y_p=-2t$.
    \ii Roots of $r^2-1$ are $\pm 1\implies y=-2t+c_1e^t+c_2e^{-t}$.
    \ii $y(0)=c_1+c_2=0$ and $y'(0)=-2+c_1-c_2=1\implies c_1=\frac32,\ c_2=-\frac32$: $y=-2t+\frac32(e^t-e^{-t})=-2t+3\sinh t$.
  \end{steps}
\end{prob}

\begin{prob}
  \textbf{\xlink{math54}{2026-11-09-undetermined-coefficients-superposition}{prob-P8}{P8}.}\par\idea{$2i$ is not a root of $r^2+1$, so $y_p=A\cos 2t+B\sin 2t$.}
  \begin{steps}
    \ii With $y_p=A\cos 2t+B\sin 2t$, $y_p''+y_p=-3A\cos 2t-3B\sin 2t$. Matching $\cos 2t$ gives $A=-\frac13$ and $B=0$.
    \ii The total is $y=-\frac13\cos 2t+c_1\cos t+c_2\sin t$. From $y(0)=-\frac13+c_1=0$ and $y'(0)=c_2=0$ we get $c_1=\frac13$, $c_2=0$, so $y=\frac13(\cos t-\cos 2t)$.
    \ii \textbf{Motion:} Superposition of two undamped oscillations at frequencies $1$ and $2$; periodic with period $2\pi$ and bounded ($|y|\le \frac23$), with no resonance since forcing frequency $2\neq 1$.
  \end{steps}
\end{prob}
